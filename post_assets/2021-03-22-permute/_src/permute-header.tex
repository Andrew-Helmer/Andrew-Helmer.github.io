  \def\pgfsysdriver{pgfsys-dvisvgm.def}

\documentclass[tikz,border=0pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\usepackage{mlmodern}

% Colors matching blog _variables.scss and tree-shuffling header palette
\definecolor{bg}{HTML}{1E1E1E}
\definecolor{treeline}{HTML}{BEBEBE}
\definecolor{mutedline}{HTML}{555555}
\definecolor{mutedbox}{HTML}{6E6E6E}
\definecolor{digitcolor}{HTML}{D9D9D9}
\definecolor{muteddigit}{HTML}{808080}
\definecolor{swapcyan}{HTML}{19D4FA}

% Closed rectangle macro using -- cycle so dvisvgm emits single-M SVG paths
\newcommand{\drawrect}[5]{
    \draw[#1] (#2, #3) -- (#4, #3) -- (#4, #5) -- (#2, #5) -- cycle;
}

\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm]
    % Uniform 0.65-unit margin around the 7-column layout (x: 3.12..15.20, y: -1.00..5.42)
    \path[use as bounding box] (2.47, -1.65) rectangle (15.85, 6.07);
    \fill[bg] (2.47, -1.65) rectangle (15.85, 6.07);

    % 7 elements (0..6) centered at x = 9.16
    \def\stepx{1.80}
    \def\topx{3.26}
    \def\arrw{1.18}
    \def\arrx{5.03}

    % Full 7-cycle bijection: 0->3, 1->6, 2->5 (cyan), 3->1, 4->0, 5->4, 6->2
    % 1. Muted background columns (0, 1, 3, 4, 5, 6)
    \foreach \i/\target in {0/3, 1/6, 3/1, 4/0, 5/4, 6/2} {
        \pgfmathsetmacro{\bx}{\topx + \i * \stepx}
        \pgfmathsetmacro{\sx}{\bx + 0.5}
        \pgfmathsetmacro{\tx}{\arrx + (\target + 0.5) * \arrw}

        % Separated input index box
        \drawrect{mutedbox, line width=1.25pt, line join=miter}{\bx}{4.42}{\bx + 1.0}{5.42}
        \node[color=muteddigit, scale=1.85, anchor=center]
            at (\sx, 4.90) {$\i$};

        % Vertical stem into isolated stateless function box
        \draw[mutedline, line width=1.3pt, -{Stealth[length=6.0pt, width=4.8pt, inset=1.2pt]}]
            (\sx, 4.35) -- (\sx, 3.36);

        % Sharp-cornered academic function box with p(i)
        \drawrect{mutedbox, line width=1.15pt, line join=miter}{\sx - 0.64}{2.50}{\sx + 0.64}{3.28}
        \node[color=muteddigit, font=\large, scale=1.22, anchor=center]
            at (\sx, 2.88) {$p(\i)$};

        % Bijection curve to target array slot
        \draw[mutedline, line width=1.3pt, -{Stealth[length=6.5pt, width=5.1pt, inset=1.3pt]}]
            (\sx, 2.43) .. controls (\sx, 1.15) and (\tx, 1.35) .. (\tx, 0.08);
    }

    % Target contiguous 7-element array at the bottom: y from -1.0 to 0.0
    \drawrect{treeline, line width=1.45pt, line join=miter}{\arrx}{-1.0}{\arrx + 7 * \arrw}{0.0}
    \foreach \i in {1,...,6} {
        \pgfmathsetmacro{\dx}{\arrx + \i * \arrw}
        \draw[treeline, line width=1.45pt] (\dx, -1.0) -- (\dx, 0.0);
    }
    % Non-highlighted target array digits (slots 0, 1, 2, 3, 4, 6)
    \foreach \i in {0, 1, 2, 3, 4, 6} {
        \pgfmathsetmacro{\cx}{\arrx + (\i + 0.5) * \arrw}
        \node[color=muteddigit, scale=1.85, anchor=center]
            at (\cx, -0.52) {$\i$};
    }
    % Highlighted target array digit (slot 5, hit by cyan query 2)
    \pgfmathsetmacro{\cxFive}{\arrx + 5.5 * \arrw}
    \node[color=digitcolor, scale=1.85, anchor=center]
        at (\cxFive, -0.52) {$5$};

    % 2. Highlighted independent query in bright cyan (column 2 -> slot 5)
    \pgfmathsetmacro{\bxTwo}{\topx + 2 * \stepx}
    \pgfmathsetmacro{\sxTwo}{\bxTwo + 0.5}
    \pgfmathsetmacro{\txFive}{\arrx + 5.5 * \arrw}

    % Highlighted input box
    \drawrect{treeline, line width=1.45pt, line join=miter}{\bxTwo}{4.42}{\bxTwo + 1.0}{5.42}
    \node[color=digitcolor, scale=1.85, anchor=center]
        at (\sxTwo, 4.90) {$2$};

    % Cyan vertical stem into isolated function box
    \draw[swapcyan, line width=1.8pt, -{Stealth[length=7.4pt, width=5.9pt, inset=1.5pt]}]
        (\sxTwo, 4.35) -- (\sxTwo, 3.36);

    % Highlighted sharp-cornered function box
    \drawrect{swapcyan, line width=1.4pt, line join=miter}{\sxTwo - 0.64}{2.50}{\sxTwo + 0.64}{3.28}
    \node[color=swapcyan, font=\large, scale=1.22, anchor=center]
        at (\sxTwo, 2.88) {$p(2)$};

    % Cyan curve to target array cell 5
    \draw[swapcyan, line width=1.95pt, -{Stealth[length=8.5pt, width=6.8pt, inset=1.7pt]}]
        (\sxTwo, 2.43) .. controls (\sxTwo, 1.15) and (\txFive, 1.35) .. (\txFive, 0.08);

\end{tikzpicture}
\end{document}
